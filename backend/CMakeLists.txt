add_library(aereborn_backend STATIC backend.cpp null_backend.cpp wav_backend.cpp wav_file.cpp)
add_library(aereborn::backend ALIAS aereborn_backend)
target_link_libraries(aereborn_backend PUBLIC aereborn::plugins PRIVATE Threads::Threads)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(ALSA QUIET)
    if(ALSA_FOUND)
        target_sources(aereborn_backend PRIVATE alsa_backend.cpp)
        target_link_libraries(aereborn_backend PRIVATE ALSA::ALSA)
        target_compile_definitions(aereborn_backend PUBLIC AEREBORN_HAS_ALSA=1)
        message(STATUS "AEReborn found the ALSA library at ${ALSA_LIBRARY}")
    else()
        message(STATUS "AEReborn skipped ALSA because its development library or headers were not found")
    endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Windows")
    target_sources(aereborn_backend PRIVATE wasapi_backend.cpp)
    target_link_libraries(aereborn_backend PRIVATE ole32 uuid)
endif()
